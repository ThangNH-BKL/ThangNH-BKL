cmake_minimum_required(VERSION 3.24)
project(CameraFtpSync VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CAMERA_BUILD_GUI "Build Windows MFC desktop application" ON)
option(CAMERA_BUILD_CURL "Build real FTP transport" ON)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/json.hpp" CAMERA_JSON_SHA256)
if(NOT CAMERA_JSON_SHA256 STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
 message(FATAL_ERROR "Vendored JSON header hash mismatch")
endif()
include(CTest)
add_library(CameraCore src/Core/Platform.cpp src/Core/Engine.cpp)
target_include_directories(CameraCore PUBLIC src/Core third_party)
target_compile_definitions(CameraCore PUBLIC UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0a00)
target_link_libraries(CameraCore PUBLIC bcrypt advapi32 shell32 ole32 uuid)
if(MSVC)
 target_compile_options(CameraCore PUBLIC /utf-8 /W4 /permissive- /EHsc)
endif()
if(CAMERA_BUILD_CURL)
 include(FetchContent)
 set(CURL_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
 set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
 set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
 set(BUILD_TESTING_SAVED ${BUILD_TESTING})
 set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
 set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
 set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
 set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
 set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
 set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
 set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
 set(HTTP_ONLY OFF CACHE BOOL "" FORCE)
 set(CURL_DISABLE_LDAP ON CACHE BOOL "" FORCE)
 set(CURL_DISABLE_LDAPS ON CACHE BOOL "" FORCE)
 set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
 set(CURL_USE_LIBSSH OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(curl URL "${CMAKE_CURRENT_SOURCE_DIR}/third_party/curl-8.12.1.tar.xz" URL_HASH SHA256=0341f1ed97a26c811abaebd37d62b833956792b7607ea3f15d001613c76de202)
 FetchContent_MakeAvailable(curl)
 set(BUILD_TESTING ${BUILD_TESTING_SAVED} CACHE BOOL "" FORCE)
 add_library(CameraTransport src/Transport/CurlFtpClient.cpp)
 target_link_libraries(CameraTransport PUBLIC CameraCore CURL::libcurl)
endif()
if(CAMERA_BUILD_GUI)
 if(NOT MSVC)
  message(FATAL_ERROR "MFC requires Visual Studio 2022 C++ and MFC x64 components. Set CAMERA_BUILD_GUI=OFF for core tests.")
 endif()
 set(CMAKE_MFC_FLAG 2)
 add_executable(CameraFtpSync WIN32 src/Gui/App.cpp src/Gui/App.rc)
 target_compile_definitions(CameraFtpSync PRIVATE _AFXDLL)
 target_link_options(CameraFtpSync PRIVATE /MANIFEST:NO)
 target_link_libraries(CameraFtpSync PRIVATE CameraTransport comctl32)
 install(TARGETS CameraFtpSync RUNTIME DESTINATION .)
 install(DIRECTORY docs DESTINATION .)
 install(FILES README.md DESTINATION .)
 install(FILES third_party/THIRD_PARTY.md third_party/CURL-LICENSE.txt third_party/JSON-LICENSE.txt third_party/SHA256SUMS.txt DESTINATION third_party)
endif()
if(BUILD_TESTING)
 add_executable(CameraCoreTests tests/CoreTests.cpp)
 target_link_libraries(CameraCoreTests PRIVATE CameraCore)
 add_test(NAME core_fault_injection COMMAND CameraCoreTests)
 add_executable(CameraResourceTests tests/Resources.cpp)
 target_link_libraries(CameraResourceTests PRIVATE CameraCore psapi)
 add_test(NAME streaming_resources COMMAND CameraResourceTests)
 if(CAMERA_BUILD_CURL)
  add_executable(CameraFtpIntegration tests/Integration.cpp)
  target_link_libraries(CameraFtpIntegration PRIVATE CameraTransport)
 endif()
endif()
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "CameraFtpSync-1.0.0-win64")
include(CPack)
